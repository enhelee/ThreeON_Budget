```python
# -*- coding: utf-8 -*-
import openpyxl

def _norm_code(code):
    if code is None:
        return None
    s = str(code).strip()
    if s.endswith(".0"):
        s = s[:-2]
    return s

class ReferenceData:
    def __init__(self, dept_by_code, attr_by_item, item_by_code):
        self._dept_by_code = dept_by_code
        self._attr_by_item = attr_by_item
        self._item_by_code = item_by_code

    @classmethod
    def load(cls, path):
        wb = openpyxl.load_workbook(path, data_only=True)
        # 부서코드: A=코드, B=부서명, C=처지사
        dept = {}
        ws = wb["부서코드"]
        for r in range(2, ws.max_row + 1):
            code = _norm_code(ws.cell(r, 1).value)
            cij = ws.cell(r, 3).value
            if code and cij:
                dept[code] = str(cij).strip()
        # 예산코드: A=예산코드, B=예산과목, E=속성
        attr, item = {}, {}
        ws = wb["예산코드"]
        for r in range(2, ws.max_row + 1):
            code = _norm_code(ws.cell(r, 1).value)
            name = ws.cell(r, 2).value
            a = ws.cell(r, 5).value
            if name:
                attr[str(name).strip()] = (str(a).strip() if a else None)
            if code and name:
                item[code] = str(name).strip()
        wb.close()
        return cls(dept, attr, item)

    def dept_of_code(self, code):
        return self._dept_by_code.get(_norm_code(code))

    def attr_of_item(self, item_name):
        if item_name is None:
            return None
        return self._attr_by_item.get(str(item_name).strip())

    def item_of_code(self, code):
        return self._item_by_code.get(_norm_code(code))
```

```python
_PL = set(config.PL_ACTUAL_ITEMS) | {d["과목"] for d in config.PL_PLAN_ITEMS}

def budget_type(item_name, ref):
    if item_name is None:
        return "기타"
    name = str(item_name).strip()
    if name in _CAP:
        return "자본"
    if name in _PL:
        return "손익"

        return "자본"
    return "기타"

def is_simui(item_name, annual_thousand):
    if str(item_name).strip() not in config.SIMUI_ITEMS:
        return False
    return float(annual_thousand or 0) >= config.SIMUI_THRESHOLD_THOUSAND
```

```python
# -*- coding: utf-8 -*-
import re
from difflib import SequenceMatcher

try:
    from rapidfuzz import fuzz as _fuzz
    def _ratio(a, b):
        return float(_fuzz.token_set_ratio(a, b))
except Exception:  # 오프라인 폴백
    def _ratio(a, b):
        return SequenceMatcher(None, a, b).ratio() * 100.0

_TOKEN = re.compile(r"[0-9A-Za-z가-힣]+")
_STOP = {"및", "등", "비용", "공사", "사업", "구매", "설치", "용역"}

def keywords(text):
    if not text:
        return set()
    toks = _TOKEN.findall(str(text))
    return {t for t in toks if len(t) >= 2 and t not in _STOP}

def _jaccard(a, b):
    ka, kb = keywords(a), keywords(b)
    if not ka or not kb:
        return 0.0
    inter = len(ka & kb)
    union = len(ka | kb)
    return 100.0 * inter / union if union else 0.0

def score(a, b):
    if not a or not b:
        return 0.0
    a, b = str(a).strip(), str(b).strip()
    if a == b:
        return 100.0
    # 문자열 유사도와 키워드 자카드의 가중 결합, 둘 중 높은 신호 반영
    r = _ratio(a, b)
    j = _jaccard(a, b)
    return max(r, 0.5 * r + 0.5 * j, j)
```

```python
# -*- coding: utf-8 -*-
from collections import defaultdict
from . import config, classifier

def aggregate_plan(df, budget="손익"):
    want = budget  # "손익" or "자본"
    matrix = defaultdict(lambda: defaultdict(float))
    for _, row in df.iterrows():
        item = row["예산과목"]
        # 분류는 이름 기반(config)만으로; ref 없이 config 목록으로 판정
        bt = "자본" if item in set(config.CAPITAL_ITEMS) else (
            "손익" if item in (set(config.PL_ACTUAL_ITEMS) | {d["과목"] for d in config.PL_PLAN_ITEMS}) else "기타")
        if bt != want:
            continue
        dept = row["처지사"]
        amt = float(row["연예산"] or 0)
        if item in config.SIMUI_ITEMS:
            key = (item, "5천만원 이상(심의대상)") if amt >= config.SIMUI_THRESHOLD_THOUSAND else (item, "5천만원 미만")
        else:
            key = (item, None)
        matrix[key][dept] += amt
    # dict화
    out = {k: dict(v) for k, v in matrix.items()}
    return {"matrix": out}

def plan_actual_rows(df):
    rows = []
    for _, row in df.iterrows():
        rows.append({
            "예산과목": row["예산과목"],
            "속성": row["속성"],
            "주관부서명": row["주관부서명"],
            "처지사": row["처지사"],
            "팀": row["부서부"],
            "사업명": row["사업명"],
            "연예산": float(row["연예산"] or 0),
        })
    return rows
```

```python
# -*- coding: utf-8 -*-
from collections import defaultdict
from . import similarity

class MatchResult:
    def __init__(self):
        self.assignments = []
        self.plan_actual_thousand = defaultdict(float)
        self.unassigned_by_group = defaultdict(float)
        self.unmatched_rows = []

def match_actuals(zrfm2_df, plan_rows, ref, threshold=80):
    res = MatchResult()
    # 계획행 인덱스: (예산과목, 처지사) → [(idx, 사업명)]
    index = defaultdict(list)
    for i, r in enumerate(plan_rows):
        index[(r["예산과목"], r["처지사"])].append((i, r["사업명"]))

    for _, z in zrfm2_df.iterrows():
        item = z["예산과목"]
        dept = ref.dept_of_code(z["부서코드P"])
        amt_thousand = float(z["금액원"] or 0) / 1000.0
        cand = index.get((item, dept), [])
        if not cand:
            res.unmatched_rows.append({
                "_orig_row": z["_orig_row"], "예산과목": item, "부서코드P": z["부서코드P"],
                "처지사": dept, "사업설명": z["사업설명"], "금액천원": amt_thousand,
                "사유": "예산과목/부서 미매칭",
            })
            res.assignments.append({"_orig_row": z["_orig_row"], "배정사업명": None,
                                    "유사도": 0, "상태": "미매칭"})
            continue
        # 3순위 사업명 유사도 최고 후보
        best_idx, best_name, best_s = None, None, -1
        for idx, name in cand:
            s = similarity.score(z["사업설명"], name)
            if s > best_s:
                best_idx, best_name, best_s = idx, name, s
        if best_s >= threshold:
            res.plan_actual_thousand[best_idx] += amt_thousand
            res.assignments.append({"_orig_row": z["_orig_row"], "배정사업명": best_name,
                                    "유사도": round(best_s, 1), "상태": "배정"})
        else:
            res.unassigned_by_group[(item, dept)] += amt_thousand
            res.assignments.append({"_orig_row": z["_orig_row"], "배정사업명": "(미지정)",
                                    "유사도": round(best_s, 1), "상태": "미지정"})
    return res
```

```python
# -*- coding: utf-8 -*-
from collections import defaultdict
from . import config

def build_actual_rows(plan_rows, match_result):
    rows = []
    for i, r in enumerate(plan_rows):
        rows.append({
            "예산과목": r["예산과목"], "속성": r["속성"], "주관부서명": r["주관부서명"],
            "처지사": r["처지사"], "팀": r["팀"], "사업명": r["사업명"],
            "연예산": r["연예산"], "실적": float(match_result.plan_actual_thousand.get(i, 0.0)),
        })
    # 그룹별 '미지정' 행 추가
    for (item, dept), amt in match_result.unassigned_by_group.items():
        rows.append({
            "예산과목": item, "속성": None, "주관부서명": None,
            "처지사": dept, "팀": None, "사업명": "(미지정)",
            "연예산": 0.0, "실적": float(amt),
        })
    return rows

def aggregate_actual(actual_rows, budget="손익"):
    cap = set(config.CAPITAL_ITEMS)
    pl = set(config.PL_ACTUAL_ITEMS)
    matrix = defaultdict(lambda: defaultdict(float))
    for r in actual_rows:
        item = r["예산과목"]
        bt = "자본" if item in cap else ("손익" if item in pl else "기타")
        if bt != budget:
            continue
        matrix[(item, None)][r["처지사"]] += float(r["실적"] or 0)
    return {"matrix": {k: dict(v) for k, v in matrix.items()}}
```

```python
# -*- coding: utf-8 -*-
def build_comparison(plan_matrix, actual_matrix):
    keys = set(plan_matrix) | set(actual_matrix)
    rows = []
    for (item, simui) in sorted(keys, key=lambda k: (str(k[0]), str(k[1]))):
        depts = set(plan_matrix.get((item, simui), {})) | set(actual_matrix.get((item, simui), {}))
        for dept in sorted(depts):
            plan = float(plan_matrix.get((item, simui), {}).get(dept, 0) or 0)
            actual = float(actual_matrix.get((item, simui), {}).get(dept, 0) or 0)
            rate = (actual / plan * 100.0) if plan else None
            rows.append({"과목": item, "심의구분": simui, "부서": dept,
                         "계획": plan, "실적": actual, "차이": actual - plan,
                         "달성률": (round(rate, 1) if rate is not None else None)})
    return rows
```

```python
from collections import Counter

def check_year(zrfm2_df, selected_year):
    years = [y for y in zrfm2_df["연도"].tolist() if y]
    if not years:
        return ["zrfm2에서 연도를 감지할 수 없습니다."]
    common = Counter(years).most_common(1)[0][0]
    if str(common) != str(selected_year):
        return [f"선택 연도({selected_year})와 zrfm2 데이터 연도({common})가 다릅니다."]
    return []

def check_depts(dept_series, ref):
    warns = []
    seen = set()
    for code in dept_series:
        if code is None or code in seen:
            continue
        seen.add(code)
        if ref.dept_of_code(code) is None:
            warns.append(f"부서코드 미매칭: {code}")
    return warns
```

```python
# -*- coding: utf-8 -*-
import openpyxl
from openpyxl.utils import get_column_letter
from . import config

def find_dept_columns(ws, header_row):
    """헤더행에서 부서명→열번호. 소계/합계/구분 제외."""
    skip = {"소 계", "소계", "합계", "합 계", "구 분", "구분", None}
    out = {}
    for c in range(1, ws.max_column + 1):
        v = ws.cell(header_row, c).value
        if v is None:
            continue
        name = str(v).strip()
        if name in skip:
            continue
        out[name] = c
    return out

def find_item_rows(ws, item_col, start_row, end_row):
    out = {}
    for r in range(start_row, end_row + 1):
        v = ws.cell(r, item_col).value
        if v is not None and str(v).strip():
            out[str(v).strip()] = r
    return out

def _write_plan_datasheet(ws, plan_df):
    # 헤더 행3 유지 가정, 4행부터 기록 (A~J)
    cols = ["주관부서명","부서코드","처지사","부서부","속성","예산코드","예산과목","사업명","산출내역","연예산"]
    r = 4
    for _, row in plan_df.iterrows():
        for i, key in enumerate(cols, 1):
            ws.cell(r, i, row[key])
        # 심의대상 자동 플래그를 별도 열(예: 26=Z)에 기록
        item = row["예산과목"]; amt = float(row["연예산"] or 0)
        flag = "O" if (item in config.SIMUI_ITEMS and amt >= config.SIMUI_THRESHOLD_THOUSAND) else ""
```

```python
# -*- coding: utf-8 -*-
import os
from . import loaders, plan_aggregator, actual_matcher, actual_aggregator
from . import comparison, validators, report, excel_writer
from .reference_data import ReferenceData

_HERE = os.path.dirname(__file__)
_APP = os.path.abspath(os.path.join(_HERE, "..", ".."))
TPL = os.path.join(_APP, "templates")
REF = os.path.join(_APP, "reference", "기준데이터.xlsx")

def run_plan(plan_path, out_dir, year):
    df = loaders.load_plan(plan_path)
    os.makedirs(out_dir, exist_ok=True)
    pl = os.path.join(out_dir, f"{year}년 손익예산_계획.xlsx")
    cap = os.path.join(out_dir, f"{year}년 자본예산_계획.xlsx")
    excel_writer.write_plan_workbook(pl, os.path.join(TPL, "손익계획_종합표.xlsx"), df, "손익")
    excel_writer.write_plan_workbook(cap, os.path.join(TPL, "자본계획_종합표.xlsx"), df, "자본")
    return {"손익계획": pl, "자본계획": cap,
            "요약": {"행수": len(df)}}

def run_actual(plan_path, zrfm2_path, out_dir, year, threshold):
    ref = ReferenceData.load(REF)
    pdf = loaders.load_plan(plan_path)
    zdf = loaders.load_zrfm2(zrfm2_path)
    prows = plan_aggregator.plan_actual_rows(pdf)
    mr = actual_matcher.match_actuals(zdf, prows, ref, threshold)
    arows = actual_aggregator.build_actual_rows(prows, mr)
    os.makedirs(out_dir, exist_ok=True)
    pl = os.path.join(out_dir, f"{year}년 손익예산_실적.xlsx")
    cap = os.path.join(out_dir, f"{year}년 자본예산_실적.xlsx")
    excel_writer.write_actual_workbook(pl, os.path.join(TPL, "손익실적_종합표.xlsx"), arows, "손익")
    excel_writer.write_actual_workbook(cap, os.path.join(TPL, "자본실적_종합표.xlsx"), arows, "자본")
    zv1 = os.path.join(out_dir, "zrfm2_V1.xlsx")
    excel_writer.write_zrfm2_v1(zv1, zrfm2_path, mr.assignments)
    # 리포트
    ywarn = validators.check_year(zdf, year)
    dwarn = validators.check_depts(zdf["부서코드P"], ref)
    from .classifier import budget_type
    unclassified = sorted({r["예산과목"] for r in prows
                           if budget_type(r["예산과목"], ref) == "기타"})
    rep_sheets = report.build_report(mr, unclassified, ywarn, dwarn, [])
    rpt = os.path.join(out_dir, "실적매칭_검토리포트.xlsx")
    excel_writer.write_generic(rpt, rep_sheets)
    # 비교표 (계획 vs 실적)
    pl_plan = plan_aggregator.aggregate_plan(pdf, "손익")["matrix"]
    cap_plan = plan_aggregator.aggregate_plan(pdf, "자본")["matrix"]
    pl_act = actual_aggregator.aggregate_actual(arows, "손익")["matrix"]
    cap_act = actual_aggregator.aggregate_actual(arows, "자본")["matrix"]
    comp = {"손익_계획대비실적": comparison.build_comparison(pl_plan, pl_act),
            "자본_계획대비실적": comparison.build_comparison(cap_plan, cap_act)}
    cmp_path = os.path.join(out_dir, "계획대비실적_비교표.xlsx")
```

```python
# -*- coding: utf-8 -*-
import os, io, zipfile, tempfile
import streamlit as st
from src.budget import pipeline

APP = os.path.dirname(os.path.abspath(__file__))
TPL = os.path.join(APP, "templates")
OUT = os.path.join(APP, "output")

def _save_upload(uploaded, suffix=".xlsx"):
    fd, path = tempfile.mkstemp(suffix=suffix)
    with os.fdopen(fd, "wb") as f:
        f.write(uploaded.getbuffer())
    return path

def _download_button(label, path):
    with open(path, "rb") as f:
        st.download_button(label, f.read(), file_name=os.path.basename(path))

def main():
    st.set_page_config(page_title="예산 자동 분석", layout="wide")
    st.title("중장기 예산 소요 전망 — 자동 분석")

    st.header("① 입력 양식 다운로드")
    c1, c2 = st.columns(2)
    with c1:
        p = os.path.join(TPL, "양식1_월별_템플릿.xlsx")
        if os.path.exists(p):
            _download_button("양식1(월별) 템플릿 다운로드", p)
    with c2:
        st.info("zrfm2는 ERP(SAP)에서 추출 후 **파일명을 반드시 `zrfm2.XLSX`** 로 저장해 업로드하세요.")

    st.header("② 분석 대상 선택")
    kind = st.radio("구분", ["계획", "실적"], horizontal=True)
    year = st.text_input("대상연도", value="2026" if kind == "계획" else "2025")

    st.header("③ 파일 업로드")
    plan_file = st.file_uploader("계획 양식1(월별) 파일", type=["xlsx"])
    zrfm2_file = None
    threshold = 80
    if kind == "실적":
        zrfm2_file = st.file_uploader("zrfm2.XLSX 파일", type=["xlsx"])
        threshold = st.slider("사업명 유사도 임계값(%)", 50, 100, 80)

    st.header("④ 분석 실행")
    if st.button("분석 실행", type="primary"):
        if not plan_file:
            st.error("계획 양식1(월별) 파일을 업로드하세요."); return
        os.makedirs(OUT, exist_ok=True)
        with st.spinner("분석 중..."):
            plan_path = _save_upload(plan_file)
            if kind == "계획":
                res = pipeline.run_plan(plan_path, OUT, year)
                st.success(f"완료: 계획 {res['요약']['행수']}행 처리")
                _download_button("손익예산_계획 다운로드", res["손익계획"])
                _download_button("자본예산_계획 다운로드", res["자본계획"])
                paths = [res["손익계획"], res["자본계획"]]
            else:
                if not zrfm2_file:
                    st.error("zrfm2 파일을 업로드하세요."); return
                if zrfm2_file.name.lower() != "zrfm2.xlsx":
                    st.warning("파일명이 zrfm2.XLSX 가 아닙니다. 계속 진행합니다.")
                zpath = _save_upload(zrfm2_file)
                res = pipeline.run_actual(plan_path, zpath, OUT, year, threshold)
                s = res["요약"]
                st.success(f"완료: 매칭 {s['매칭']} / 미지정 {s['미지정']} / 미매칭 {s['미매칭']}")
                for w in s["연도경고"]:
                    st.warning(w)
                for k in ("손익실적","자본실적","zrfm2_V1","검토리포트","비교표"):
                    _download_button(f"{k} 다운로드", res[k])
                paths = [res[k] for k in ("손익실적","자본실적","zrfm2_V1","검토리포트","비교표")]
        # 전체 ZIP
        buf = io.BytesIO()
        with zipfile.ZipFile(buf, "w") as z:
            for p in paths:
                z.write(p, os.path.basename(p))
        st.download_button("전체 ZIP 다운로드", buf.getvalue(), file_name=f"{year}_{kind}_결과.zip")

if __name__ == "__main__":
    main()
```
