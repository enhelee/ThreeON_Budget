# 연도별 기준정보 DB화 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 지사 구성·과목 구성·기준정보 마스터를 연도별로 DB에 저장하고, 설정 화면에서 편집하며, 마감 해제 구간을 승인과 배너로 감싼다.

**Architecture:** 연도마다 완전한 한 벌을 저장하는 다섯 개 표(`dept_config`·`item_config`·`item_master`·`dept_master`·`alias`)를 만든다. `config_store`의 파일 IO를 DB 조회로 바꾸되 함수 이름과 반환 형태는 유지해, 호출부는 인자만 `config_dir` → `conn`으로 바뀐다. 초기값은 연도별 코드 시드로 심어 파일이 없는 환경에서도 첫 기동만으로 옳은 값이 된다.

**Tech Stack:** Python 3.14 · FastAPI · pandas · SQLite/PostgreSQL(dbcore 전환계층) · Vite + Vanilla JS SPA · pytest

**Spec:** [docs/superpowers/specs/2026-09-15-연도별-기준정보-DB화-design.md](../specs/2026-09-15-연도별-기준정보-DB화-design.md)

```python
def migrate_masters_to_years(conn):
    """연도 없이 쌓인 마스터 행을 데이터가 있는 모든 연도로 복제한다.

    한 번만 의미가 있다 — year 가 NULL 인 행이 없으면 아무것도 하지 않는다.
    연도를 알 수 없는 행이므로 «어느 해 것인지» 고를 근거가 없고, 그래서
    전 연도에 같은 내용을 둔다. 이관 직후 동작이 바뀌지 않는 쪽을 택한 것이다.
    """
    years = [r[0] for r in conn.execute(
        "SELECT DISTINCT year FROM dataset ORDER BY year")]
    if not years:
        return
    for table in ("item_master", "dept_master"):
        legacy = conn.execute(
            f"SELECT * FROM {table} WHERE year IS NULL").fetchall()
        if not legacy:
            continue
        cols = [d[0] for d in conn.execute(f"SELECT * FROM {table} LIMIT 0").description]
        others = [c for c in cols if c != "year"]
        ph = ",".join("?" for _ in others) + ",?"
        for y in years:
            for row in legacy:
                vals = [row[cols.index(c)] for c in others] + [y]
                conn.execute(
                    f"INSERT INTO {table}({','.join(others)},year) VALUES({ph})", vals)
        conn.execute(f"DELETE FROM {table} WHERE year IS NULL")
    conn.commit()
```

```python
def _seed_year(year):
    """시드에 없는 연도는 가장 가까운 과거 연도를 쓴다(없으면 가장 이른 연도)."""
    years = sorted(SEED_DEPTS_BY_YEAR)
    past = [y for y in years if y <= str(year)]
    return past[-1] if past else years[0]


def seed_dept_config(year):
    src = SEED_DEPTS_BY_YEAR[_seed_year(year)]
    out = []
    for g in DEPT_GROUPS:
        for name in src[g]:
            out.append({"이름": name, "그룹": g, "포함": True})
    return out
```

```python
    rows = conn.execute(
        "SELECT 과목,대분류,심의대상,포함,실적반영 FROM item_config"

        save_item_config(conn, year, budget, seeded)

             "포함": bool(r[3]), "실적반영": bool(r[4])} for r in rows]


def save_item_config(conn, year, budget, items):
    items = [normalize_item_row(x) for x in items]
    conn.execute("DELETE FROM item_config WHERE year=? AND budget=?",
                 (str(year), budget))
    for i, x in enumerate(items):
        conn.execute(
            "INSERT INTO item_config(year,budget,과목,대분류,심의대상,포함,실적반영,순서)"
            " VALUES(?,?,?,?,?,?,?,?)",
            (str(year), budget, x["과목"], x.get("대분류"),
             1 if to_bool(x.get("심의대상"), False) else 0,
             1 if to_bool(x.get("포함"), True) else 0,
             1 if to_bool(x.get("실적반영"), True) else 0, i))
    conn.commit()
    return items


def _load_alias(conn, 종류, seed):
    merged = dict(seed)
    for a, b in conn.execute(
            "SELECT 원표기,정규표기 FROM alias WHERE 종류=?", (종류,)).fetchall():
        merged[a] = b
    return merged

    conn.execute("DELETE FROM alias WHERE 종류=?", (종류,))
    for a, b in mapping.items():
        conn.execute("INSERT INTO alias(종류,원표기,정규표기) VALUES(?,?,?)",
                     (종류, a, b))
```
